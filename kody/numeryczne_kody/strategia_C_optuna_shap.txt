import os
import pickle
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import matplotlib
matplotlib.use('Agg')
import time

from sklearn.compose import TransformedTargetRegressor
from sklearn.ensemble import GradientBoostingRegressor, RandomForestRegressor
from sklearn.linear_model import Ridge
from sklearn.neural_network import MLPRegressor
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import LeaveOneGroupOut, cross_val_score
from sklearn.metrics import r2_score, mean_absolute_error
import xgboost as xgb
import optuna
import shap

optuna.logging.set_verbosity(optuna.logging.WARNING)

MERGED_CSV = "ADI_dataset_v8.csv"
SAVE_DIR   = "tabular_results"
os.makedirs(SAVE_DIR, exist_ok=True)

TARGETS = ['Rm_MPa', 'Rp02_MPa', 'A_pct', 'impact_J', 'HB']
TAB_COLS = [
    'grubosc_mm', 'T_aust_C', 't_aust_min', 'T_hart_C', 't_hart_min',
    'chem_C', 'chem_Si', 'chem_Mn', 'chem_S', 'chem_P',
    'chem_Mg', 'chem_Cu', 'chem_Ni'
]

N_TRIALS      = 20
INNER_SCORING = 'neg_mean_absolute_error'


def build_model(model_name: str, params: dict):
    if model_name == 'GBT':
        return GradientBoostingRegressor(**params)
    elif model_name == 'XGBoost':
        return xgb.XGBRegressor(**params)
    elif model_name == 'RandomForest':
        return RandomForestRegressor(**params)
    elif model_name == 'Ridge':
        return TransformedTargetRegressor(
            regressor=Ridge(**params),
            transformer=StandardScaler(),
        )
    elif model_name == 'MLP':
        return TransformedTargetRegressor(
            regressor=MLPRegressor(**params),
            transformer=StandardScaler(),
        )
    raise ValueError(f"Nieznany model: {model_name}")


def suggest_optuna_params(trial, model_name: str) -> dict:
    if model_name == 'GBT':
        return dict(
            n_estimators     = trial.suggest_int('n_estimators', 100, 2000),
            max_depth        = trial.suggest_int('max_depth', 2, 5),
            learning_rate    = trial.suggest_float('learning_rate', 0.005, 0.30, log=True),
            subsample        = trial.suggest_float('subsample', 0.6, 1.0),
            min_samples_leaf = trial.suggest_int('min_samples_leaf', 1, 10),
            random_state=42,
        )
    elif model_name == 'XGBoost':
        return dict(
            n_estimators     = trial.suggest_int('n_estimators', 400, 4000),
            max_depth        = trial.suggest_int('max_depth', 2, 6),
            learning_rate    = trial.suggest_float('learning_rate', 0.005, 0.30, log=True),
            subsample        = trial.suggest_float('subsample', 0.5, 1.0),
            colsample_bytree = trial.suggest_float('colsample_bytree', 0.5, 1.0),
            reg_alpha        = trial.suggest_float('reg_alpha', 0.01, 10.0, log=True),
            reg_lambda       = trial.suggest_float('reg_lambda', 0.01, 10.0, log=True),
            random_state=42, verbosity=0,
        )
    elif model_name == 'RandomForest':
        return dict(
            n_estimators     = trial.suggest_int('n_estimators', 100, 500),
            max_depth        = trial.suggest_int('max_depth', 2, 15),
            max_features     = trial.suggest_float('max_features', 0.3, 1.0),
            min_samples_leaf = trial.suggest_int('min_samples_leaf', 1, 8),
            random_state=42, n_jobs=-1,
        )
    elif model_name == 'Ridge':
        return dict(alpha=trial.suggest_float('alpha', 1e-3, 1e3, log=True))
    elif model_name == 'MLP':
        n_layers = trial.suggest_int('n_layers', 1, 3)
        layers   = tuple(trial.suggest_int(f'units_{i}', 16, 128)
                         for i in range(n_layers))
        return dict(
            hidden_layer_sizes = layers,
            alpha              = trial.suggest_float('alpha', 1e-5, 1e-1, log=True),
            learning_rate_init = trial.suggest_float('lr', 1e-4, 1e-1, log=True),
            max_iter=500, random_state=42,
            early_stopping=True, validation_fraction=0.15,
        )
    raise ValueError(f"Nieznany model: {model_name}")


def rebuild_from_optuna_best(model_name: str, best_params: dict):
    return build_model(
        model_name,
        suggest_optuna_params(optuna.trial.FixedTrial(best_params), model_name),
    )


df = pd.read_csv(MERGED_CSV)


def extract_folder_name(x):
    x = str(x).strip().replace('\\\\', '\\').replace('\\', '/')
    parts = [p for p in x.split('/') if p]
    return parts[-1] if parts else x


df['folder_key'] = df['folder_1000x'].apply(extract_folder_name)
available_tab    = [c for c in TAB_COLS if c in df.columns]
agg_dict         = {col: 'first' for col in available_tab + TARGETS
                    if col in df.columns}
df_agg = (df.groupby('folder_key')
            .agg(agg_dict)
            .reset_index()
            .reset_index(drop=True))

print(f"Próbek: {len(df_agg)}")
print(f"Cechy ({len(available_tab)}): {available_tab}\n")

logo_outer  = LeaveOneGroupOut()
all_results = []
MODEL_NAMES = ['GBT', 'XGBoost', 'RandomForest', 'Ridge', 'MLP']
shap_data_per_target = {}

for target in TARGETS:
    sub = df_agg.dropna(subset=[target]).reset_index(drop=True)
    X   = sub[available_tab].fillna(sub[available_tab].median()).values
    y   = sub[target].values
    grp = np.arange(len(sub))

    print(f"\n{'='*70}")
    print(f"TARGET: {target}  ({len(sub)} próbek)")

    shap_X_all, shap_y_true_all = [], []
    shap_folders, gbt_models_all = [], []

    for model_name in MODEL_NAMES:
        y_true_all, y_pred_all = [], []
        t0 = time.time()

        for fold_idx, (train_idx, test_idx) in enumerate(
                logo_outer.split(sub, y, groups=grp)):

            scaler  = StandardScaler()
            X_train = scaler.fit_transform(X[train_idx])
            X_test  = scaler.transform(X[test_idx])
            y_train = y[train_idx]

            inner_logo   = LeaveOneGroupOut()
            inner_groups = np.arange(len(train_idx))

            def objective(trial):
                m = build_model(
                    model_name,
                    suggest_optuna_params(trial, model_name),
                )
                scores = cross_val_score(
                    m, X_train, y_train,
                    cv=inner_logo,
                    groups=inner_groups,
                    scoring=INNER_SCORING,
                    n_jobs=-1,
                )
                return scores.mean()

            study = optuna.create_study(
                direction='maximize',
                sampler=optuna.samplers.TPESampler(seed=42),
            )
            study.optimize(objective, n_trials=N_TRIALS, show_progress_bar=False)

            model = rebuild_from_optuna_best(model_name, study.best_params)
            model.fit(X_train, y_train)
            pred = float(model.predict(X_test)[0])

            if model_name == 'GBT':
                shap_X_all.append(X_test[0])
                shap_y_true_all.append(float(y[test_idx[0]]))
                shap_folders.append(sub.loc[test_idx[0], 'folder_key'])
                gbt_core = (model.regressor_
                            if hasattr(model, 'regressor_')
                            else model)
                gbt_models_all.append(gbt_core)

            y_true_all.append(float(y[test_idx[0]]))
            y_pred_all.append(pred)

        elapsed = time.time() - t0
        r2  = r2_score(np.array(y_true_all), np.array(y_pred_all))
        mae = mean_absolute_error(np.array(y_true_all), np.array(y_pred_all))
        all_results.append({
            'target':    target,
            'model':     model_name,
            'strategy':  'C_optuna',
            'n_samples': len(sub),
            'R2':        round(r2, 3),
            'MAE':       round(mae, 3),
            'time_s':    round(elapsed, 1),
        })
        print(f"  {model_name:15s} [C_optuna]  "
              f"R²={r2:.3f}  MAE={mae:.3f}  ({elapsed:.0f}s)")

    shap_data_per_target[target] = {
        'X_all':         np.stack(shap_X_all) if shap_X_all else None,
        'y_true':        np.array(shap_y_true_all),
        'folders':       shap_folders,
        'gbt_models':    gbt_models_all,
        'feature_names': available_tab,
    }

df_results = pd.DataFrame(all_results)
df_results.to_csv(
    os.path.join(SAVE_DIR, "tabular_strategy_C_optuna.csv"), index=False
)

print("\n" + "="*70)
print("ZESTAWIENIE R² — Strategia C (Optuna TPE, inner LOGO)")
print("="*70)
pivot = df_results.groupby(['target', 'model'])['R2'].max().unstack()
print(pivot.to_string())

shap_save_dir = os.path.join(SAVE_DIR, "shap")
os.makedirs(shap_save_dir, exist_ok=True)

for target in TARGETS:
    d = shap_data_per_target[target]
    if d['X_all'] is None:
        continue

    X_all         = d['X_all']
    feature_names = d['feature_names']
    gbt_models    = d['gbt_models']

    shap_values_all = []
    for i, gbt_model in enumerate(gbt_models):
        explainer = shap.TreeExplainer(gbt_model)
        sv = explainer.shap_values(X_all[i].reshape(1, -1))
        shap_values_all.append(sv[0])

    shap_matrix = np.stack(shap_values_all)

    shap.summary_plot(shap_matrix, X_all,
                      feature_names=feature_names,
                      show=False, plot_type="dot")
    plt.title(f"SHAP Summary — GBT (Optuna+inner LOGO) — {target}",
              fontsize=13, pad=12)
    plt.tight_layout()
    plt.savefig(
        os.path.join(shap_save_dir, f"shap_summary_{target}.png"),
        dpi=150, bbox_inches='tight',
    )
    plt.close()

    mean_abs_shap = np.abs(shap_matrix).mean(axis=0)
    sorted_idx    = np.argsort(mean_abs_shap)[::-1]

    fig, ax = plt.subplots(figsize=(9, 5))
    ax.barh(
        [feature_names[i] for i in sorted_idx],
        mean_abs_shap[sorted_idx],
        color='steelblue',
    )
    ax.set_xlabel("Mean |SHAP value|")
    ax.set_title(
        f"Ważność cech (SHAP) — GBT (Optuna+inner LOGO) — {target}",
        fontsize=13,
    )
    ax.invert_yaxis()
    plt.tight_layout()
    plt.savefig(
        os.path.join(shap_save_dir, f"shap_bar_{target}.png"),
        dpi=150, bbox_inches='tight',
    )
    plt.close()

    np.savez(
        os.path.join(shap_save_dir, f"shap_values_{target}.npz"),
        shap_values=shap_matrix,
        X=X_all,
        feature_names=np.array(feature_names),
        folders=np.array(d['folders']),
    )

    print(f"  SHAP {target} — Top 3: "
          + ", ".join([f"{feature_names[i]} ({mean_abs_shap[i]:.3f})"
                       for i in sorted_idx[:3]]))

    with open(os.path.join(shap_save_dir, f"gbt_final_{target}.pkl"), 'wb') as f:
        pickle.dump(gbt_models[-1], f)

print(f"\n{'='*70}")
print(f"Zapisano w: {SAVE_DIR}/")
