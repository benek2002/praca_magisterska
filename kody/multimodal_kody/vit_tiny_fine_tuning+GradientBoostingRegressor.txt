def fold_done_path(target, fold_idx):
    return os.path.join(
        PROGRESS_DIR,
        f"{target}_fold{fold_idx + 1}.json"
    )


def save_fold(target, fold_idx, data):
    serializable = {}

    for key, value in data.items():
        if isinstance(value, np.ndarray):
            serializable[key] = value.tolist()
        else:
            serializable[key] = value

    with open(
        fold_done_path(target, fold_idx),
        "w"
    ) as file:
        json.dump(serializable, file)


def load_fold(target, fold_idx):
    path = fold_done_path(target, fold_idx)

    if not os.path.exists(path):
        return None

    with open(path) as file:
        data = json.load(file)

    for key in ["shap_values", "x_test", "embedding"]:
        if key in data and data[key] is not None:
            data[key] = np.array(data[key])

    return data

class MultiStopDataset(Dataset):
    def __init__(self, path_label_pairs, transform):
        self.pairs = path_label_pairs
        self.transform = transform

    def __len__(self):
        return len(self.pairs)

    def __getitem__(self, index):
        path, label = self.pairs[index]

        image = Image.open(path).convert("RGB")
        image = self.transform(image)

        label = torch.tensor(
            float(label),
            dtype=torch.float32
        )

        return image, label


class StopDataset(Dataset):
    def __init__(self, paths, label_norm, transform):
        self.paths = paths
        self.label_norm = float(label_norm)
        self.transform = transform

    def __len__(self):
        return len(self.paths)

    def __getitem__(self, index):
        image = Image.open(
            self.paths[index]
        ).convert("RGB")

        image = self.transform(image)

        label = torch.tensor(
            self.label_norm,
            dtype=torch.float32
        )

        return image, label

def extract_folder_name(value):
    return (
        str(value)
        .strip()
        .replace("\\", "/")
        .split("/")[-1]
    )


def get_image_paths(folder):
    paths = []

    for extension in IMG_EXTS:
        paths.extend(
            glob.glob(
                os.path.join(folder, extension)
            )
        )

    return sorted(paths)


def unwrap(model):
    if isinstance(model, nn.DataParallel):
        return model.module

    return model

def build_vit_model(device, verbose=False):
    model = timm.create_model(
        VIT_NAME,
        pretrained=True,
        num_classes=0
    )

    for parameter in model.parameters():
        parameter.requires_grad = False

    for parameter in model.blocks[-1].parameters():
        parameter.requires_grad = True

    for parameter in model.norm.parameters():
        parameter.requires_grad = True

    model.head = nn.Sequential(
        nn.Linear(EMBED_DIM, 64),
        nn.ReLU(),
        nn.Dropout(0.3),
        nn.Linear(64, 1),
    )

    if verbose:
        trainable = sum(
            parameter.numel()
            for parameter in model.parameters()
            if parameter.requires_grad
        )

        total = sum(
            parameter.numel()
            for parameter in model.parameters()
        )

        print(
            f"  Params: {trainable:,} / {total:,} "
            f"({100 * trainable / total:.1f}% trainable)"
        )

    return model.to(device)


def extract_embeddings(model, image_paths, device):
    base = unwrap(model)
    base.eval()

    embeddings = []

    with torch.no_grad():
        for start in range(
            0,
            len(image_paths),
            BATCH_SIZE
        ):
            batch_paths = image_paths[
                start:start + BATCH_SIZE
            ]

            images = torch.stack([
                val_transform(
                    Image.open(path).convert("RGB")
                )
                for path in batch_paths
            ]).to(device)

            x = base.patch_embed(images)

            cls_token = base.cls_token.expand(
                x.shape[0],
                -1,
                -1
            )

            x = torch.cat(
                (cls_token, x),
                dim=1
            )

            x = base.pos_drop(
                x + base.pos_embed
            )

            for block in base.blocks:
                x = block(x)

            x = base.norm(x)

            embeddings.append(
                x[:, 0].cpu().numpy()
            )

    return np.concatenate(
        embeddings,
        axis=0
    ).mean(axis=0)

        np.abs(shap_values[N_TAB:]).sum()

        SHAP_DIR,

        target_dir,
        exist_ok=True

        target_dir,
        f"fold{fold_idx + 1}_{stop[:18]}"
    )

    fig, ax = plt.subplots(
        figsize=(5, 3)

    bars = ax.bar(
        [
            f"Obraz\n(PCA {N_TAB}D)",
            f"Tabela\n(raw {N_TAB}D)"

        color=["steelblue", "darkorange"],
        edgecolor="k"

        f"Tabela={100 * tabular_abs / total:.1f}%\n"

def build_train_loader(
    sub,
    indices,
    normalize_target
):
    pairs = []

    for index in indices:
        image_paths = get_image_paths(
            sub.loc[index, "folder_path"]
        )

        label = normalize_target(
            sub.loc[index, "target_value"]
        )

        for path in image_paths:
            pairs.append(
                (path, label)
            )

    if not pairs:
        return None

    return DataLoader(
        MultiStopDataset(
            pairs,
            train_transform
        ),
        batch_size=BATCH_SIZE,
        shuffle=True,
        num_workers=4,
        pin_memory=True,
        persistent_workers=True,
    )


def build_val_loader(
    sub,
    index,
    normalize_target
):
    image_paths = get_image_paths(
        sub.loc[index, "folder_path"]
    )

    if not image_paths:
        return None

    label = normalize_target(
        sub.loc[index, "target_value"]
    )

    return DataLoader(
        StopDataset(
            image_paths,
            label,
            val_transform
        ),
        batch_size=BATCH_SIZE,
        shuffle=False,
        num_workers=4,
        pin_memory=True,
        persistent_workers=True,
    )

def train_vit(
    model,
    train_loader,
    val_loader,
    y_mean,
    y_std
):
    base = unwrap(model)

    trainable_parameters = [
        parameter
        for parameter in model.parameters()
        if parameter.requires_grad
    ]

    optimizer = optim.Adam([
        {
            "params": base.blocks[-1].parameters(),
            "lr": LR * 0.1
        },
        {
            "params": base.norm.parameters(),
            "lr": LR * 0.1
        },
        {
            "params": base.head.parameters(),
            "lr": LR
        },
    ], weight_decay=1e-4)

    scheduler = optim.lr_scheduler.CosineAnnealingLR(
        optimizer,
        T_max=EPOCHS,
        eta_min=LR * 0.05
    )

    criterion = nn.SmoothL1Loss()

    best_val_loss = float("inf")
    best_state = None
    patience_counter = 0

    for epoch in range(EPOCHS):
        model.train()

        for images, labels in train_loader:
            images = images.to(
                DEVICE,
                non_blocking=True
            )

            labels = labels.to(
                DEVICE,
                non_blocking=True
            ).unsqueeze(1)

            optimizer.zero_grad()

            predictions = model(images)
            loss = criterion(
                predictions,
                labels
            )

            loss.backward()

            nn.utils.clip_grad_norm_(
                trainable_parameters,
                1.0
            )

            optimizer.step()

        scheduler.step()

        model.eval()

        validation_losses = []

        with torch.no_grad():
            for images, labels in val_loader:
                images = images.to(
                    DEVICE,
                    non_blocking=True
                )

                labels = labels.to(
                    DEVICE,
                    non_blocking=True
                ).unsqueeze(1)

                predictions = model(images)

                validation_losses.append(
                    criterion(
                        predictions,
                        labels
                    ).item()
                )

        val_loss = float(
            np.mean(validation_losses)
        )

        if val_loss < best_val_loss - 1e-4:
            best_val_loss = val_loss

            best_state = {
                key: value.clone()
                for key, value
                in model.state_dict().items()
            }

            patience_counter = 0
        else:
            patience_counter += 1

            if patience_counter >= PATIENCE:
                print(
                    f"    Early stop @ epoch {epoch + 1}"
                )
                break

    if best_state is not None:
        model.load_state_dict(
            best_state
        )

        # ----------------------------------------------------
        # TARGET NORMALIZATION
        # ----------------------------------------------------

        y_train = y[train_idx]

        y_mean = float(
            y_train.mean()
        )

        y_std = float(
            y_train.std()
        ) + 1e-8

        def normalize_target(value):
            return (
                float(value) - y_mean
            ) / y_std

        # ----------------------------------------------------
        # TRAIN / VALIDATION SPLIT
        # ----------------------------------------------------

        rng = np.random.RandomState(
            SEED + fold_idx
        )

        validation_position = int(
            rng.choice(
                len(train_idx)
            )
        )

        validation_index = train_idx[
            validation_position
        ]

        pure_train_idx = np.delete(
            train_idx,
            validation_position
        )

        train_loader = build_train_loader(
            sub,
            pure_train_idx,
            normalize_target
        )

        val_loader = build_val_loader(
            sub,
            validation_index,
            normalize_target
        )

        if train_loader is None:
            print(
                "  SKIP — brak zdjęć"
            )
            continue

        # ----------------------------------------------------
        # VIT TRAINING
        # ----------------------------------------------------

        model = build_vit_model(
            DEVICE,
            verbose=(fold_idx == 0)
        )

        train_vit(
            model,
            train_loader,
            val_loader,
            y_mean,
            y_std
        )

        # ----------------------------------------------------
        # EMBEDDINGS
        # ----------------------------------------------------

        embeddings = {}

        extraction_indices = (
            list(train_idx)
            + [test_index]
        )

        for index in extraction_indices:
            image_paths = get_image_paths(
                sub.loc[
                    index,
                    "folder_path"
                ]
            )

            if image_paths:
                embeddings[index] = (
                    extract_embeddings(
                        model,
                        image_paths,
                        DEVICE
                    )
                )
            else:
                embeddings[index] = np.zeros(
                    EMBED_DIM
                )

        E_train = np.stack([
            embeddings[index]
            for index in train_idx
        ])

        E_test = embeddings[
            test_index
        ].reshape(1, -1)

        # ----------------------------------------------------
        # MODEL CHECKPOINT
        # ----------------------------------------------------

        checkpoint_path = os.path.join(
            CKPT_DIR,
            f"{target}_fold{fold_idx + 1}.pth"
        )

        torch.save(
            {
                "state_dict": model.state_dict(),
                "target": target,
                "fold": fold_idx + 1,
                "stop": test_stop,
                "y_mean": y_mean,
                "y_std": y_std,
            },
            checkpoint_path
        )

        del model
        del train_loader
        del val_loader

        torch.cuda.empty_cache()
        gc.collect()

        # ----------------------------------------------------
        # TABULAR FEATURES
        # ----------------------------------------------------

        tabular_raw = (
            sub[TABULAR_COLS]
            .copy()
        )

        train_median = (
            tabular_raw
            .iloc[train_idx]
            .median()
        )

        tabular_raw = (
            tabular_raw
            .fillna(train_median)
        )

        scaler = StandardScaler()

        X_tab_train = (
            scaler
            .fit_transform(
                tabular_raw
                .iloc[train_idx]
                .values
            )
            .astype(np.float32)
        )

        X_tab_test = (
            scaler
            .transform(
                tabular_raw
                .iloc[[test_index]]
                .values
            )
            .astype(np.float32)
        )

        # ----------------------------------------------------
        # PCA
        # ----------------------------------------------------

        n_components = min(
            N_TAB,
            len(train_idx) - 1
        )

        pca = PCA(
            n_components=n_components,
            random_state=SEED
        )

        E_train_pca = (
            pca.fit_transform(
                E_train
            )
        )

        E_test_pca = (
            pca.transform(
                E_test
            )
        )

        if n_components < N_TAB:
            padding = N_TAB - n_components

            E_train_pca = np.pad(
                E_train_pca,
                ((0, 0), (0, padding))
            )

            E_test_pca = np.pad(
                E_test_pca,
                ((0, 0), (0, padding))
            )

        explained_variance = float(
            pca
            .explained_variance_ratio_
            .sum()
            * 100
        )

        # ----------------------------------------------------
        # FINAL FEATURES
        # ----------------------------------------------------

        X_train = np.concatenate(
            [
                E_train_pca,
                X_tab_train
            ],
            axis=1
        )

        X_test = np.concatenate(
            [
                E_test_pca,
                X_tab_test
            ],
            axis=1
        )

        # ----------------------------------------------------
        # GBT
        # ----------------------------------------------------

        gbt = GradientBoostingRegressor(
            random_state=SEED
        )

        gbt.fit(
            X_train,
            y[train_idx]
        )

        predicted_value = float(
            gbt.predict(X_test)[0]
        )

        true_value = float(
            y[test_index]
        )

        absolute_error = abs(
            true_value - predicted_value
        )

        relative_error = (
            absolute_error
            / (abs(true_value) + 1e-8)
            * 100
        )

        print(
            f"  ⇒ true={true_value:7.1f} "
            f"pred={predicted_value:7.1f} "
            f"|err|={absolute_error:6.1f} "
            f"({relative_error:.1f}%) "
            f"PCA_var={explained_variance:.1f}%"
        )

        # ----------------------------------------------------
        # SHAP
        # ----------------------------------------------------

        explainer = shap.TreeExplainer(
            gbt
        )

        shap_values = (
            explainer
            .shap_values(X_test)[0]
        )

        image_abs, tabular_abs = (
            plot_shap_fold(
                shap_values,
                target,
                fold_idx,
                test_stop,
                true_value,
                predicted_value
            )
        )

        total_shap = (
            image_abs
            + tabular_abs
            + 1e-8
        )

        shap_image_pct = round(
            100 * image_abs / total_shap,
            1
        )

        shap_tabular_pct = round(
            100 * tabular_abs / total_shap,
            1
        )

        shap_values_all.append(
            shap_values
        )

        x_test_all.append(
            X_test[0]
        )

        modal_rows.append({
            "shap_img_pct": shap_image_pct,
            "shap_tab_pct": shap_tabular_pct,
        })

        # ----------------------------------------------------
        # SAVE FOLD
        # ----------------------------------------------------

        save_fold(
            target,
            fold_idx,
            {
                "true": true_value,
                "pred": predicted_value,
                "stop": test_stop,
                "n_imgs": len(test_images),
                "pca_var_pct": round(
                    explained_variance,
                    1
                ),
                "shap_img_pct": shap_image_pct,
                "shap_tab_pct": shap_tabular_pct,
                "shap_values": shap_values,
                "x_test": X_test[0],
            }
        )

        y_true_all.append(
            true_value
        )

        y_pred_all.append(
            predicted_value
        )

        detail_rows.append({
            "target": target,
            "fold": fold_idx + 1,
            "stop": test_stop,
            "n_imgs": len(test_images),
            "true": round(
                true_value,
                3
            ),
            "pred": round(
                predicted_value,
                3
            ),
            "abs_err": round(
                absolute_error,
                3
            ),
            "rel_err_pct": round(
                relative_error,
                2
            ),
            "pca_var_pct": round(
                explained_variance,
                1
            ),
        })

    # ========================================================
    # TARGET RESULTS
    # ========================================================

    if len(y_true_all) < 2:
        continue

    y_true = np.array(
        y_true_all
    )

    y_pred = np.array(
        y_pred_all
    )

    r2 = r2_score(
        y_true,
        y_pred
    )

    mae = mean_absolute_error(
        y_true,
        y_pred
    )

    rmse = float(
        np.sqrt(
            np.mean(
                (y_true - y_pred) ** 2
            )
        )
    )

    mape = float(
        np.mean(
            np.abs(
                (y_true - y_pred)
                / (np.abs(y_true) + 1e-8)
            )
        )
        * 100
    )

    all_results.append({
        "target": target,
        "model": (
            f"ViT-Tiny_unfrozen_"
            f"PCA{N_TAB}_GBT_default"
        ),
        "n_stops": len(y_true),
        "R2": round(r2, 4),
        "MAE": round(mae, 3),
        "RMSE": round(rmse, 3),
        "MAPE_pct": round(mape, 2),
    })

    print(
        f"\n  >>> {target} "
        f"R²={r2:.3f} "
        f"MAE={mae:.3f} "
        f"RMSE={rmse:.3f} "
        f"MAPE={mape:.1f}%"
    )

    if len(shap_values_all) >= 2:
        plot_shap_summary(
            shap_values_all,
            x_test_all,
            target,
            modal_rows
        )

    # ========================================================
    # SCATTER PLOT
    # ========================================================

    fig, ax = plt.subplots(
        figsize=(5, 5)
    )

    ax.scatter(
        y_true,
        y_pred,
        color="mediumseagreen",
        edgecolors="k",
        s=70,
        zorder=3
    )

    limits = [
        min(
            y_true.min(),
            y_pred.min()
        ) * 0.95,
        max(
            y_true.max(),
            y_pred.max()
        ) * 1.05
    ]

    ax.plot(
        limits,
        limits,
        "r--",
        lw=1.5,
        label="ideał"
    )

    ax.set_xlabel(
        f"Wartość rzeczywista [{target}]"
    )

    ax.set_ylabel(
        f"Predykcja [{target}]"
    )

    ax.set_title(
        f"ViT-Tiny unfrozen+Tab→GBT default\n"
        f"{target}  "
        f"R²={r2:.3f}  "
        f"MAE={mae:.1f}"
    )

    ax.legend()
    ax.grid(
        True,
        alpha=0.3
    )

    plt.tight_layout()

    plt.savefig(
        os.path.join(
            RESULTS_DIR,
            f"scatter_{target}.png"
        ),
        dpi=150
    )

    plt.close()
