SEED = 42


def set_seed(s):
    random.seed(s)
    np.random.seed(s)
    torch.manual_seed(s)
    torch.cuda.manual_seed_all(s)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False


set_seed(SEED)

def extract_folder_name(x):
    return (
        str(x)
        .strip()
        .replace("\\", "/")
        .split("/")[-1]
    )


def get_image_paths(folder_path):
    paths = []

    for ext in IMG_EXTS:
        paths += glob.glob(
            os.path.join(folder_path, ext)
        )

    return sorted(paths)


class StopDataset(Dataset):
    def __init__(
        self,
        paths,
        label_norm,
        transform
    ):
        self.paths = paths
        self.label_norm = float(label_norm)
        self.transform = transform

    def __len__(self):
        return len(self.paths)

    def __getitem__(self, idx):
        img = Image.open(
            self.paths[idx]
        ).convert("RGB")

        return (
            self.transform(img),
            torch.tensor(
                self.label_norm,
                dtype=torch.float32
            )
        )


class MultiStopDataset(Dataset):
    def __init__(
        self,
        path_label_pairs,
        transform
    ):
        self.pairs = path_label_pairs
        self.transform = transform

    def __len__(self):
        return len(self.pairs)

    def __getitem__(self, idx):
        path, label = self.pairs[idx]

        img = Image.open(path).convert("RGB")

        return (
            self.transform(img),
            torch.tensor(
                float(label),
                dtype=torch.float32
            )
        )


train_transform = transforms.Compose([
    transforms.Resize(
        (IMG_SIZE, IMG_SIZE)
    ),
    transforms.RandomHorizontalFlip(),
    transforms.RandomVerticalFlip(),
    transforms.RandomRotation(15),
    transforms.RandomAffine(
        degrees=0,
        shear=8
    ),
    transforms.ColorJitter(
        brightness=0.2,
        contrast=0.2,
        saturation=0.1
    ),
    transforms.GaussianBlur(
        kernel_size=3,
        sigma=(0.1, 1.0)
    ),
    transforms.ToTensor(),
    transforms.Normalize(
        [0.485, 0.456, 0.406],
        [0.229, 0.224, 0.225]
    ),
])


val_transform = transforms.Compose([
    transforms.Resize(
        (IMG_SIZE, IMG_SIZE)
    ),
    transforms.ToTensor(),
    transforms.Normalize(
        [0.485, 0.456, 0.406],
        [0.229, 0.224, 0.225]
    ),
])


def build_model(
    device,
    verbose=False
):
    base = models.resnet18(
        weights=models.ResNet18_Weights.IMAGENET1K_V1
    )

    for p in base.parameters():
        p.requires_grad = False

    for p in base.layer4.parameters():
        p.requires_grad = True

    base.fc = nn.Sequential(
        nn.Linear(
            base.fc.in_features,
            128
        ),
        nn.ReLU(),
        nn.Dropout(0.3),
        nn.Linear(128, 1),
    )

    if verbose:
        trainable = sum(
            p.numel()
            for p in base.parameters()
            if p.requires_grad
        )

        total = sum(
            p.numel()
            for p in base.parameters()
        )

        print(
            f"Parametry: {trainable:,} / "
            f"{total:,} "
            f"({100 * trainable / total:.1f}% trainable)"
        )

    if USE_DDP:
        base = nn.DataParallel(base)

    return base.to(device)


def unwrap(model):
    if isinstance(
        model,
        nn.DataParallel
    ):
        return model.module

    return model


class GradCAM:
    def __init__(self, model):
        self.model = unwrap(model)
        self.activations = None
        self.gradients = None

        target = self.model.layer4[-1].conv2

        self._fh = target.register_forward_hook(
            self._save_act
        )

        self._bh = target.register_full_backward_hook(
            self._save_grad
        )

    def _save_act(
        self,
        module,
        inputs,
        output
    ):
        self.activations = output

    def _save_grad(
        self,
        module,
        grad_input,
        grad_output
    ):
        self.gradients = grad_output[0]

    def compute(self, img_tensor):
        self.model.eval()
        self.model.zero_grad()

        with torch.enable_grad():
            out = self.model(img_tensor)
            out = out.mean()
            out.backward()

        grads = self.gradients
        acts = self.activations

        weights = grads.mean(
            dim=(2, 3),
            keepdim=True
        )

        cam = (
            weights * acts
        ).sum(dim=1).squeeze(0)

        cam = F.relu(
            cam
        ).detach().cpu().numpy()

        cam -= cam.min()

        if cam.max() > 0:
            cam /= cam.max()

        return cam

    def remove_hooks(self):
        self._fh.remove()
        self._bh.remove()


def save_gradcam_figure(
    img_path,
    cam,
    pred_val,
    true_val,
    save_path,
    alpha=0.5
):
    img = cv2.imread(img_path)

    img = cv2.cvtColor(
        cv2.resize(
            img,
            (IMG_SIZE, IMG_SIZE)
        ),
        cv2.COLOR_BGR2RGB
    )

    cam_r = cv2.resize(
        cam,
        (IMG_SIZE, IMG_SIZE)
    )

    heatmap = (
        cm.jet(cam_r)[:, :, :3] * 255
    ).astype(np.uint8)

    overlay = (
        alpha * heatmap
        + (1 - alpha) * img
    ).astype(np.uint8)

    fig, axes = plt.subplots(
        1,
        3,
        figsize=(13, 4)
    )

    axes[0].imshow(img)
    axes[0].set_title("Oryginał")
    axes[0].axis("off")

    axes[1].imshow(
        cam_r,
        cmap="jet"
    )
    axes[1].set_title("GradCAM")
    axes[1].axis("off")

    axes[2].imshow(overlay)
    axes[2].set_title(
        f"Overlay\n"
        f"true={true_val:.1f} "
        f"pred={pred_val:.1f}"
    )
    axes[2].axis("off")

    plt.tight_layout()

    plt.savefig(
        save_path,
        dpi=120,
        bbox_inches="tight"
    )

    plt.close()


df = pd.read_csv(CSV_PATH)

df["folder_key"] = df[
    "folder_1000x"
].apply(
    extract_folder_name
)

agg_dict = {
    t: "first"
    for t in TARGETS
}

df_agg = (
    df.groupby("folder_key")
    .agg(agg_dict)
    .reset_index()
)

df_agg["folder_path"] = df_agg[
    "folder_key"
].apply(
    lambda x: os.path.join(
        ROOT_DIR,
        x
    )
)

df_agg = df_agg[
    df_agg["folder_path"].apply(
        os.path.isdir
    )
].reset_index(drop=True)


print(
    f"Stopów: {len(df_agg)}"
)

for _, row in df_agg.iterrows():
    n = len(
        get_image_paths(
            row["folder_path"]
        )
    )

    print(
        f"{row['folder_key']:35s} | "
        f"zdjęć={n:3d}"
    )


logo = LeaveOneGroupOut()

all_results = []
detail_rows = []


for target in TARGETS:

    sub = (
        df_agg
        .dropna(subset=[target])
        .reset_index(drop=True)
    )

    grp = np.arange(
        len(sub)
    )

    y = sub[
        target
    ].values.astype(
        np.float32
    )

    print(
        f"\nTARGET: {target} "
        f"({len(sub)} stopów)"
    )

    build_model(
        DEVICE,
        verbose=True
    )

    if len(sub) < 3:
        print(
            "Za mało danych, pomijam"
        )
        continue

    y_true_all = []
    y_pred_all = []

    for fold_idx, (
        train_idx,
        test_idx
    ) in enumerate(
        logo.split(
            sub,
            y,
            groups=grp
        )
    ):
        test_i = int(
            test_idx[0]
        )

        test_folder = sub.loc[
            test_i,
            "folder_key"
        ]

        y_train = y[
            train_idx
        ]

        y_mean = float(
            y_train.mean()
        )

        y_std = (
            float(y_train.std())
            + 1e-8
        )

        norm = lambda v: (
            float(v) - y_mean
        ) / y_std

        denorm = lambda v: (
            float(v) * y_std
            + y_mean
        )

        rng_val = np.random.RandomState(
            SEED + fold_idx
        )

        val_local = int(
            rng_val.choice(
                len(train_idx)
            )
        )

        val_i = train_idx[
            val_local
        ]

        pure_train = np.delete(
            train_idx,
            val_local
        )

        def build_train_dl(indices):
            pairs = []

            for i in indices:
                for p in get_image_paths(
                    sub.loc[
                        i,
                        "folder_path"
                    ]
                ):
                    pairs.append(
                        (
                            p,
                            norm(y[i])
                        )
                    )

            if not pairs:
                return None

            ds = MultiStopDataset(
                pairs,
                train_transform
            )

            return DataLoader(
                ds,
                batch_size=BATCH_SIZE,
                shuffle=True,
                num_workers=4,
                pin_memory=True,
                persistent_workers=True
            )

        def build_val_dl(i):
            paths = get_image_paths(
                sub.loc[
                    i,
                    "folder_path"
                ]
            )

            if not paths:
                return None

            ds = StopDataset(
                paths,
                norm(y[i]),
                val_transform
            )

            return DataLoader(
                ds,
                batch_size=BATCH_SIZE,
                shuffle=False,
                num_workers=4,
                pin_memory=True,
                persistent_workers=True
            )

        train_dl = build_train_dl(
            pure_train
        )

        val_dl = build_val_dl(
            val_i
        )

        test_dl = build_val_dl(
            test_i
        )

        if (
            train_dl is None
            or test_dl is None
        ):
            print(
                f"Fold {fold_idx + 1:2d} "
                f"[{test_folder}] - SKIP"
            )
            continue

        model = build_model(
            DEVICE
        )

        trainable = [
            p
            for p in model.parameters()
            if p.requires_grad
        ]

        base = unwrap(model)

        optimizer = optim.Adam(
            [
                {
                    "params":
                        base.layer4.parameters(),
                    "lr":
                        LR * 0.1
                },
                {
                    "params":
                        base.fc.parameters(),
                    "lr":
                        LR
                },
            ],
            weight_decay=1e-4
        )

        scheduler = (
            optim.lr_scheduler.CosineAnnealingLR(
                optimizer,
                T_max=EPOCHS,
                eta_min=LR * 0.05
            )
        )

        criterion = nn.SmoothL1Loss()

        best_val = float("inf")
        best_state = None
        patience_cnt = 0

        for epoch in range(EPOCHS):
            model.train()

            for imgs, lbls in train_dl:
                imgs = imgs.to(
                    DEVICE,
                    non_blocking=True
                )

                lbls = lbls.to(
                    DEVICE,
                    non_blocking=True
                ).unsqueeze(1)

                optimizer.zero_grad()

                loss = criterion(
                    model(imgs),
                    lbls
                )

                loss.backward()

                nn.utils.clip_grad_norm_(
                    trainable,
                    1.0
                )

                optimizer.step()

            scheduler.step()

            if val_dl:
                model.eval()

                vl = []

                with torch.no_grad():
                    for imgs, lbls in val_dl:
                        imgs = imgs.to(
                            DEVICE,
                            non_blocking=True
                        )

                        lbls = lbls.to(
                            DEVICE,
                            non_blocking=True
                        ).unsqueeze(1)

                        vl.append(
                            criterion(
                                model(imgs),
                                lbls
                            ).item()
                        )

                val_loss = float(
                    np.mean(vl)
                )

                if val_loss < (
                    best_val - 1e-4
                ):
                    best_val = val_loss

                    best_state = {
                        k: v.clone()
                        for k, v
                        in model.state_dict().items()
                    }

                    patience_cnt = 0

                else:
                    patience_cnt += 1

                    if patience_cnt >= PATIENCE:
                        print(
                            f"Early stop @ "
                            f"epoch {epoch + 1}"
                        )
                        break

        if best_state:
            model.load_state_dict(
                best_state
            )

        model.eval()

        preds_norm = []

        with torch.no_grad():
            for imgs, _ in test_dl:
                imgs = imgs.to(
                    DEVICE,
                    non_blocking=True
                )

                out = model(
                    imgs
                ).squeeze(1).cpu().numpy()

                preds_norm.extend(
                    out.tolist()
                )

        pred_real = denorm(
            float(
                np.mean(preds_norm)
            )
        )

        true_real = float(
            y[test_i]
        )

        y_true_all.append(
            true_real
        )

        y_pred_all.append(
            pred_real
        )

        err_abs = abs(
            true_real - pred_real
        )

        err_pct = (
            err_abs
            / (abs(true_real) + 1e-8)
            * 100
        )

        print(
            f"Fold {fold_idx + 1:2d} "
            f"[{test_folder:30s}] "
            f"true={true_real:7.1f} "
            f"pred={pred_real:7.1f} "
            f"|err|={err_abs:6.1f} "
            f"({err_pct:.1f}%)"
        )

        detail_rows.append({
            "target": target,
            "fold": fold_idx + 1,
            "stop": test_folder,
            "true": round(
                true_real,
                3
            ),
            "pred": round(
                pred_real,
                3
            ),
            "abs_err": round(
                err_abs,
                3
            ),
            "rel_err_pct": round(
                err_pct,
                2
            ),
        })

        test_paths = get_image_paths(
            sub.loc[
                test_i,
                "folder_path"
            ]
        )

        per_img_preds = []

        gcam = GradCAM(
            model
        )

        for img_path in test_paths:
            img_t = val_transform(
                Image.open(
                    img_path
                ).convert("RGB")
            ).unsqueeze(0).to(DEVICE)

            with torch.no_grad():
                pred_single = (
                    unwrap(model)(
                        img_t
                    ).item()
                )

            per_img_preds.append(
                (
                    img_path,
                    pred_single
                )
            )

        pred_mean_norm = float(
            np.mean(preds_norm)
        )

        per_img_preds.sort(
            key=lambda x:
                abs(
                    x[1]
                    - pred_mean_norm
                )
        )

        top_paths = per_img_preds[
            :GRADCAM_TOPK
        ]

        fold_gcam_dir = os.path.join(
            GRADCAM_DIR,
            target,
            f"fold{fold_idx + 1}_"
            f"{test_folder}"
        )

        os.makedirs(
            fold_gcam_dir,
            exist_ok=True
        )

        for rank, (
            img_path,
            img_pred_norm
        ) in enumerate(top_paths):

            img_t = val_transform(
                Image.open(
                    img_path
                ).convert("RGB")
            ).unsqueeze(0).to(DEVICE)

            cam = gcam.compute(
                img_t
            )

            img_pred_real = denorm(
                img_pred_norm
            )

            fname = os.path.basename(
                img_path
            )

            save_gradcam_figure(
                img_path,
                cam,
                pred_val=img_pred_real,
                true_val=true_real,
                save_path=os.path.join(
                    fold_gcam_dir,
                    f"top{rank + 1}_{fname}.png"
                ),
            )

        gcam.remove_hooks()

        torch.save(
            {
                "state_dict":
                    model.state_dict(),
                "target":
                    target,
                "fold":
                    fold_idx + 1,
                "stop":
                    test_folder,
                "y_mean":
                    y_mean,
                "y_std":
                    y_std,
            },
            os.path.join(
                CKPT_DIR,
                f"{target}_fold"
                f"{fold_idx + 1}.pth"
            )
        )

        del (
            model,
            train_dl,
            val_dl,
            test_dl
        )

        torch.cuda.empty_cache()
        gc.collect()

    if len(y_true_all) < 2:
        continue

    y_t = np.array(
        y_true_all
    )

    y_p = np.array(
        y_pred_all
    )

    r2 = r2_score(
        y_t,
        y_p
    )

    mae = mean_absolute_error(
        y_t,
        y_p
    )

    rmse = float(
        np.sqrt(
            np.mean(
                (y_t - y_p) ** 2
            )
        )
    )

    mape = float(
        np.mean(
            np.abs(
                (y_t - y_p)
                / (np.abs(y_t) + 1e-8)
            )
        ) * 100
    )

    all_results.append({
        "target": target,
        "n_stops": len(
            y_true_all
        ),
        "R2": round(
            r2,
            4
        ),
        "MAE": round(
            mae,
            3
        ),
        "RMSE": round(
            rmse,
            3
        ),
        "MAPE_pct": round(
            mape,
            2
        ),
    })

    print(
        f"{target} "
        f"R²={r2:.3f} "
        f"MAE={mae:.3f} "
        f"RMSE={rmse:.3f} "
        f"MAPE={mape:.1f}%"
    )

    fig, ax = plt.subplots(
        figsize=(5, 5)
    )

    ax.scatter(
        y_t,
        y_p,
        color="steelblue",
        edgecolors="k",
        s=70,
        zorder=3
    )

    lim = [
        min(
            y_t.min(),
            y_p.min()
        ) * 0.95,
        max(
            y_t.max(),
            y_p.max()
        ) * 1.05
    ]

    ax.plot(
        lim,
        lim,
        "r--",
        lw=1.5,
        label="ideał"
    )

    ax.set_xlabel(
        f"Wartość rzeczywista [{target}]"
    )

    ax.set_ylabel(
        f"Predykcja [{target}]"
    )

    ax.set_title(
        f"{target} | "
        f"R²={r2:.3f} "
        f"MAE={mae:.1f}"
    )

    ax.legend()
    ax.grid(
        True,
        alpha=0.3
    )

    plt.tight_layout()

    plt.savefig(
        os.path.join(
            RESULTS_DIR,
            f"scatter_{target}.png"
        ),
        dpi=150
    )

    plt.close()


summary_df = pd.DataFrame(
    all_results
)

summary_df.to_csv(
    os.path.join(
        RESULTS_DIR,
        "summary.csv"
    ),
    index=False
)

pd.DataFrame(
    detail_rows
).to_csv(
    os.path.join(
        RESULTS_DIR,
        "per_fold.csv"
    ),
    index=False
)

print(
    summary_df[
        ["target", "R2", "MAE"]
    ].to_string(
        index=False
    )
)

print(
    f"Wyniki: {RESULTS_DIR}"
)
